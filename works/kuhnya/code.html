<script>
  document.querySelectorAll<HTMLFormElement>('[data-quiz]').forEach((form) => {
    const steps = [...form.querySelectorAll<HTMLFieldSetElement>('[data-step]')];
    const bar = form.querySelector<HTMLElement>('[data-bar]')!;
    const count = form.querySelector<HTMLElement>('[data-count]')!;
    const nav = form.querySelector<HTMLElement>('[data-nav]')!;
    const answers = form.querySelector<HTMLInputElement>('[name=answers]')!;
    let i = 0;

    form.classList.add('is-js');
    nav.hidden = false;

    const go = (n: number) => {
      i = Math.max(0, Math.min(steps.length - 1, n));
      steps.forEach((s, k) => { s.hidden = k !== i; });
      bar.style.width = `${((i + 1) / steps.length) * 100}%`;
      count.textContent = `Шаг ${i + 1} из ${steps.length}`;
      form.querySelector<HTMLElement>('[data-back]')!.style.visibility = i ? 'visible' : 'hidden';
      // Собираем ответы в одно поле — так они придут менеджеру одной строкой
      const data = new FormData(form);
      answers.value = ['Планировка', 'Размер', 'Стиль', 'Сроки', 'Связь']
        .map((k) => data.get(k) && `${k}: ${data.get(k)}`).filter(Boolean).join('; ');
    };

    // Выбор варианта сразу переводит на следующий шаг — меньше кликов, выше конверсия
    steps.slice(0, -1).forEach((s, k) => s.addEventListener('change', () => setTimeout(() => go(k + 1), 220)));
    form.querySelector('[data-back]')!.addEventListener('click', () => go(i - 1));
    form.addEventListener('change', () => go(i));
    go(0);
  });
</script>
