<script>
(() => {
  "use strict";

  /* ——— цветовые преобразования ——— */
  const hexToRgb = (hex) => {
    const h = hex.replace("#", "").trim();
    const full = h.length === 3 ? h.split("").map(c => c + c).join("") : h;
    const n = parseInt(full, 16);
    return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
  };
  const rgbToHex = (rgb) =>
    "#" + rgb.map(v => Math.round(Math.min(255, Math.max(0, v))).toString(16).padStart(2, "0")).join("").toUpperCase();

  const toLinear = (c) => { c /= 255; return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4); };
  const fromLinear = (c) => 255 * (c <= 0.0031308 ? c * 12.92 : 1.055 * Math.pow(c, 1 / 2.4) - 0.055);

  function rgbToOklab([r, g, b]) {
    const R = toLinear(r), G = toLinear(g), B = toLinear(b);
    const l = Math.cbrt(0.4122214708 * R + 0.5363325363 * G + 0.0514459929 * B);
    const m = Math.cbrt(0.2119034982 * R + 0.6806995451 * G + 0.1073969566 * B);
    const s = Math.cbrt(0.0883024619 * R + 0.2817188376 * G + 0.6299787005 * B);
    return [
      0.2104542553 * l + 0.7936177850 * m - 0.0040720468 * s,
      1.9779984951 * l - 2.4285922050 * m + 0.4505937099 * s,
      0.0259040371 * l + 0.7827717662 * m - 0.8086757660 * s
    ];
  }

  function oklabToRgb([L, a, b]) {
    const l_ = L + 0.3963377774 * a + 0.2158037573 * b;
    const m_ = L - 0.1055613458 * a - 0.0638541728 * b;
    const s_ = L - 0.0894841775 * a - 1.2914855480 * b;
    const l = l_ ** 3, m = m_ ** 3, s = s_ ** 3;
    return [
      fromLinear(4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s),
      fromLinear(-1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s),
      fromLinear(-0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s)
    ];
  }

  const oklabToOklch = ([L, a, b]) => [L, Math.hypot(a, b), Math.atan2(b, a) * 180 / Math.PI];
  const oklchToOklab = ([L, C, h]) => [L, C * Math.cos(h * Math.PI / 180), C * Math.sin(h * Math.PI / 180)];

  const inGamut = (rgb) => rgb.every(v => v >= -0.6 && v <= 255.6);

  // Гамут-мэппинг: тянем хрому вниз, пока цвет не влезет в sRGB
  function mapToGamut([L, C, h]) {
    let c = C;
    for (let i = 0; i < 200; i++) {
      const rgb = oklabToRgb(oklchToOklab([L, c, h]));
      if (inGamut(rgb)) return oklchToOklab([L, c, h]);
      c *= 0.985;
      if (c < 0.0005) break;
    }
    return [L, 0, 0];
  }

  const lum = (rgb) => {
    const [r, g, b] = rgb.map(toLinear);
    return 0.2126 * r + 0.7152 * g + 0.0722 * b;
  };
  const contrast = (a, b) => {
    const l1 = lum(hexToRgb(a)), l2 = lum(hexToRgb(b));
    return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05);
  };

  /* ——— шкала ——— */
  const STEPS = [
    { k: 50, L: 0.985, c: 0.10 },
    { k: 100, L: 0.955, c: 0.19 },
    { k: 200, L: 0.905, c: 0.34 },
    { k: 300, L: 0.835, c: 0.56 },
    { k: 400, L: 0.745, c: 0.80 },
    { k: 500, L: 0.655, c: 1.00 },
    { k: 600, L: 0.575, c: 0.98 },
    { k: 700, L: 0.495, c: 0.88 },
    { k: 800, L: 0.415, c: 0.74 },
    { k: 900, L: 0.335, c: 0.60 },
    { k: 950, L: 0.245, c: 0.46 }
  ];

  const state = { base: "#4F46E5", steps: [], fg: "#111111", bg: "#FFFFFF", target: "fg" };

  function buildSteps(hex) {
    const [, C0, h] = oklabToOklch(rgbToOklab(hexToRgb(hex)));
    return STEPS.map(s => {
      const lab = mapToGamut([s.L, C0 * s.c, h]);
      const rgb = oklabToRgb(lab);
      const [L, C, hh] = oklabToOklch(lab);
      return {
        k: s.k,
        hex: rgbToHex(rgb),
        oklch: `oklch(${(L * 100).toFixed(1)}% ${C.toFixed(3)} ${hh.toFixed(0)})`
      };
    });
  }

  function renderScale() {
    const wrap = document.getElementById("scale");
    wrap.innerHTML = state.steps.map(s => {
      const dark = lum(hexToRgb(s.hex)) < 0.35;
      return `<button class="step" type="button" data-hex="${s.hex}" data-active="${s.hex === (state.target === "fg" ? state.fg : state.bg) ? 1 : 0}"
        style="background:${s.hex};color:${dark ? "rgba(255,255,255,.8)" : "rgba(0,0,0,.6)"}"
        title="oklch ${s.k} · ${s.hex}" aria-label="Ступень ${s.k}, ${s.hex}"><em>${s.k}</em><b>${s.hex.slice(1)}</b></button>`;
    }).join("");

    wrap.querySelectorAll(".step").forEach(btn => {
      btn.addEventListener("click", () => {
        if (state.target === "fg") state.fg = btn.dataset.hex; else state.bg = btn.dataset.hex;
        renderScale(); renderContrast();
      });
    });

    const meta = document.getElementById("base-meta");
    const [, C, h] = oklabToOklch(rgbToOklab(hexToRgb(state.base)));
    meta.innerHTML = `OKLCH<br><b>${state.base}</b><br>h ${h.toFixed(0)}° · C ${C.toFixed(3)}`;

    document.getElementById("r-fg").style.background = state.fg;
    document.getElementById("r-bg").style.background = state.bg;
    document.getElementById("r-acc").style.background = state.steps[5].hex;
  }

  function renderContrast() {
    const ratio = contrast(state.fg, state.bg);
    const r = ratio.toFixed(2).replace(".", ",");
    document.getElementById("ratio").textContent = r;
    document.getElementById("pv-big").style.color = state.fg;
    document.getElementById("pv-small").style.color = state.fg;
    document.getElementById("preview").style.background = state.bg;
    document.getElementById("pick-fg").innerHTML = `текст<b>${state.fg}</b>`;
    document.getElementById("pick-bg").innerHTML = `фон<b>${state.bg}</b>`;
    document.getElementById("pick-fg").setAttribute("aria-pressed", String(state.target === "fg"));
    document.getElementById("pick-bg").setAttribute("aria-pressed", String(state.target === "bg"));
    document.getElementById("pick-fg").style.boxShadow = state.target === "fg" ? "inset 0 0 0 1px #fff" : "none";
    document.getElementById("pick-bg").style.boxShadow = state.target === "bg" ? "inset 0 0 0 1px #fff" : "none";

    const mark = (id, ok) => {
      const el = document.getElementById(id);
      el.textContent = ok ? "проходит" : "ниже нормы";
      el.className = ok ? "pass" : "fail";
    };
    mark("b-aa", ratio >= 4.5);
    mark("b-aaa", ratio >= 7);
    mark("b-aal", ratio >= 3);
    mark("b-ui", ratio >= 3);

    document.getElementById("css-out").value = cssVars();
  }

  const cssVars = () => {
    const lines = state.steps.map(s => `  --c-${s.k}: ${s.hex.toLowerCase()}; /* ${s.oklch} */`);
    lines.push(`  --c-base: ${state.base.toLowerCase()};`);
    lines.push(`  --c-ink: ${state.fg.toLowerCase()};`);
    lines.push(`  --c-paper: ${state.bg.toLowerCase()};`);
    return `:root {\n${lines.join("\n")}\n}`;
  };

  function setBase(hex) {
    const clean = (hex.startsWith("#") ? hex : "#" + hex).toUpperCase();
    if (!/^#[0-9A-F]{6}$/.test(clean)) return;
    state.base = clean;
    document.getElementById("picker").value = clean.toLowerCase();
    document.getElementById("hexin").value = clean.slice(1);
    state.steps = buildSteps(clean);
    // по умолчанию: тёмная и светлая ступени как текст/фон
    state.fg = state.steps[8].hex;
    state.bg = state.steps[0].hex;
    renderScale(); renderContrast();
  }

  /* ——— события ——— */
  document.getElementById("picker").addEventListener("input", (e) => setBase(e.target.value));
  document.getElementById("hexin").addEventListener("change", (e) => {
    const v = e.target.value.replace(/[^0-9a-f]/gi, "");
    if (v.length === 6) setBase(v); else e.target.value = state.base.slice(1);
  });
  document.getElementById("pick-fg").addEventListener("click", () => { state.target = "fg"; renderScale(); renderContrast(); });
  document.getElementById("pick-bg").addEventListener("click", () => { state.target = "bg"; renderScale(); renderContrast(); });

  document.getElementById("swap").addEventListener("click", () => {
    [state.fg, state.bg] = [state.bg, state.fg];
    renderScale(); renderContrast();
  });

  // ищем ближайшую по свету ступень, дающую нужный контраст
  document.getElementById("fix").addEventListener("click", () => {
    const candidates = state.steps.map(s => s.hex);
    const want = contrast(state.fg, state.bg) >= 4.5
      ? null
      : candidates.map(hx => ({ hx, c: contrast(hx, state.bg) })).sort((a, b) => b.c - a.c)[0];
    if (want && want.c > 0) {
      state.fg = want.hx;
      document.getElementById("out-hint").textContent = `Подставил ${want.hx}: ${want.c.toFixed(2).replace(".", ",")}:1 к фону.`;
    } else {
      document.getElementById("out-hint").textContent = "Контраст уже выше 4,5:1 — менять нечего.";
    }
    renderScale(); renderContrast();
  });

  async function copyText(text, btn, label) {
    try {
      await navigator.clipboard.writeText(text);
      btn.dataset.done = "1";
      const old = btn.textContent;
      btn.textContent = "Готово";
      setTimeout(() => { btn.textContent = old; delete btn.dataset.done; }, 1500);
    } catch {
      document.getElementById("out-hint").textContent = "Буфер обмена закрыт — скопируйте из поля ниже.";
    }
  }

  document.getElementById("copy-css").addEventListener("click", (e) => copyText(cssVars(), e.currentTarget));
  document.getElementById("copy-hex").addEventListener("click", (e) => copyText(state.steps[5].hex, e.currentTarget));

  setBase("#4F46E5");
})();
</script>
