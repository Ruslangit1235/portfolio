<script>
(() => {
  "use strict";
  const stage = document.getElementById("stage");
  const sample = document.getElementById("sample");
  const grid = document.getElementById("glyph-grid");
  const famLabel = document.getElementById("glyph-fam");
  const outSize = document.getElementById("o-size");
  const outTrack = document.getElementById("o-track");
  const outLh = document.getElementById("o-lh");

  const state = { family: "Unbounded", weight: 400, size: 64, track: -0.02, lh: 1 };

  const GLYPHS = "АБВГДЕЁЖЗИЙКЛМНОПРСТУФХЦЧШЩЪЫЬЭЮЯабвгдеёжзийклмнопрстуфхцчшщъыьэюя0123456789.,:;!?—–()«»№%";

  function apply() {
    sample.style.fontFamily = `"${state.family}", sans-serif`;
    sample.style.fontWeight = state.weight;
    sample.style.fontSize = state.size + "px";
    sample.style.letterSpacing = state.track + "em";
    sample.style.lineHeight = state.lh;
    // авто-высота textarea: рамка растёт под набранное
    sample.style.height = "auto";
    sample.style.height = Math.max(state.size * state.lh, sample.scrollHeight) + "px";
    stage.dataset.meta = `${state.family} ${state.weight} · ${state.size}px`;
    famLabel.textContent = `${state.family} · ${state.weight}`;
    outSize.value = state.size;
    outTrack.value = state.track.toFixed(2).replace(".", ",");
    outLh.value = state.lh.toFixed(2).replace(".", ",");
    renderGlyphs();
  }

  function renderGlyphs() {
    const chars = GLYPHS.split("");
    // перерисовываем только при смене гарнитуры — экономим layout
    if (grid.dataset.sig === state.family + state.weight) return;
    grid.dataset.sig = state.family + state.weight;
    grid.innerHTML = chars.map(c => `<span style="font-family:'${state.family}',sans-serif;font-weight:${state.weight}">${c}</span>`).join("");
  }

  document.querySelectorAll("[data-family]").forEach((btn) => {
    btn.addEventListener("click", () => {
      state.family = btn.dataset.family;
      document.querySelectorAll("[data-family]").forEach(b => b.setAttribute("aria-pressed", String(b === btn)));
      apply();
    });
  });

  document.querySelectorAll("[data-weight]").forEach((btn) => {
    btn.addEventListener("click", () => {
      state.weight = +btn.dataset.weight;
      document.querySelectorAll("[data-weight]").forEach(b => b.setAttribute("aria-pressed", String(b === btn)));
      apply();
    });
  });

  document.getElementById("f-size").addEventListener("input", (e) => { state.size = +e.target.value; apply(); });
  document.getElementById("f-track").addEventListener("input", (e) => { state.track = +e.target.value / 1000; apply(); });
  document.getElementById("f-lh").addEventListener("input", (e) => { state.lh = +e.target.value / 100; apply(); });
  sample.addEventListener("input", apply);

  // карточки каталога переключают тестер
  document.querySelectorAll("[data-set]").forEach((card) => {
    card.addEventListener("click", () => {
      const [family, weight] = card.dataset.set.split("|");
      state.family = family;
      state.weight = +weight;
      document.querySelectorAll("[data-family]").forEach(b => b.setAttribute("aria-pressed", String(b.dataset.family === family)));
      document.querySelectorAll("[data-weight]").forEach(b => b.setAttribute("aria-pressed", String(+b.dataset.weight === +weight)));
      apply();
      document.getElementById("tester").scrollIntoView({ behavior: "smooth", block: "start" });
      setTimeout(() => sample.focus({ preventScroll: true }), 400);
    });
  });

  const copyBtn = document.getElementById("copy-css");
  copyBtn.addEventListener("click", async () => {
    const css = `font-family: "${state.family}", sans-serif;\nfont-weight: ${state.weight};\nfont-size: ${state.size}px;\nletter-spacing: ${state.track.toFixed(3)}em;\nline-height: ${state.lh.toFixed(2)};`;
    try {
      await navigator.clipboard.writeText(css);
      copyBtn.textContent = "Скопировано";
      copyBtn.dataset.done = "1";
    } catch {
      copyBtn.textContent = "Не вышло — выделите вручную";
    }
    setTimeout(() => { copyBtn.textContent = "Скопировать CSS"; delete copyBtn.dataset.done; }, 1800);
  });

  apply();
})();
</script>
