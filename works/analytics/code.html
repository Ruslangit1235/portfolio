      const weekend = dow === 5 || dow === 6 ? 0.72 : 1;
      const trend = 1 + i * 0.004;
      const visits = Math.round(base * weekend * trend * (0.9 + rnd() * 0.2));
      const uniq = Math.round(visits * (0.62 + rnd() * 0.1));
      const regs = Math.round(visits * (0.055 + rnd() * 0.02));
      out.push({ visits, uniq, regs });

    traffic: ["Продукт / трафик", "Трафик и устройства"],
    funnel: ["Продукт / воронка", "Воронка продукта"],
    reports: ["Продукт / отчёты", "Отчёты и выгрузка"],
  };

  function show(view) {
    state.view = view;
    $$(".view").forEach((v) => (v.hidden = v.id !== "view-" + view));
    $$(".rail button").forEach((b) =>
      b.setAttribute("aria-current", b.dataset.view === view ? "page" : "false")
    );
    const [crumb, title] = TITLES[view];
    $("#crumb").textContent = crumb;
    $("#title").textContent = title;
    $("#range-chips").style.visibility = view === "reports" ? "hidden" : "visible";

    // новые размеры контейнеров появляются только после показа
    requestAnimationFrame(() => {
      if (view === "overview") renderOverview();
      if (view === "traffic") renderTraffic();
      if (view === "funnel") renderFunnel();
      if (view === "reports") renderReports();
    });
  }

  $$(".rail button").forEach((btn) => {
    btn.addEventListener("click", () => show(btn.dataset.view));
  });

  /* ——— период и обновление ——— */
  $$("[data-range]").forEach((btn) => {
    btn.addEventListener("click", () => {
      state.range = +btn.dataset.range;
      $$("[data-range]").forEach((b) => b.setAttribute("aria-pressed", String(b === btn)));
      state.series = buildSeries(state.range);
      show(state.view);
    });
  });

  $("#shuffle").addEventListener("click", () => {
    seed += 7919;
    state.series = buildSeries(state.range);
    show(state.view);
    toast("Данные пересчитаны");
  });

  $("#make-report").addEventListener("click", () => {
    const now = new Date();
    reports.unshift({
      name: "Сводка по продукту",
      period: state.range + " дн",
      rows: Math.round(state.range * (38 + rnd() * 20)),
      at: now.toLocaleDateString("ru-RU", { day: "2-digit", month: "2-digit" }) + ", " +
          now.toLocaleTimeString("ru-RU", { hour: "2-digit", minute: "2-digit" }),
    });
    renderReports();
    toast("Отчёт сформирован — лежит первым в списке");
  });

  $("#download-all").addEventListener("click", (e) => {
    download("skala-vse-otchyoty.csv", csvFor("90 дн"));
    const b = e.currentTarget;
    b.dataset.done = "1";
    b.textContent = "Скачано";
    setTimeout(() => { b.textContent = "Скачать всё одним CSV"; delete b.dataset.done; }, 1600);
  });

  /* ——— часы ——— */
  const clock = $("#live-clock");
  const tick = () => (clock.textContent = new Date().toLocaleTimeString("ru-RU"));
  tick(); setInterval(tick, 1000);

  /* ——— старт ——— */
  state.series = buildSeries(state.range);
  show("overview");
})();
</script>
